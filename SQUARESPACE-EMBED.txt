<iframe
  id="braves-dash-frame"
  src="https://jasonbhorne.github.io/braves-dashboard/"
  title="Braves True-Talent Dashboard"
  style="width:1px;min-width:100%;border:0;display:block;min-height:800px;"
  scrolling="no"
  loading="lazy"></iframe>
<script>
  window.addEventListener("message", function (e) {
    if (e.data && e.data.bravesDashboard && e.data.height) {
      document.getElementById("braves-dash-frame").style.height =
        e.data.height + "px";
    }
  });
</script>
